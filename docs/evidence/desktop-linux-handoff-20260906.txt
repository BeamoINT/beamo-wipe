import json, os, pathlib, struct, subprocess, time, urllib.request, urllib.parse

def marker(text):
    with open('/dev/ttyS0','w') as f: f.write(text+'\n')

def run(*args): return subprocess.check_output(args, text=True).strip()

try:
    assert 'boot=live' not in pathlib.Path('/proc/cmdline').read_text().split()
    disks=json.loads(run('/usr/bin/lsblk','--json','--tree','--bytes','--output','PATH,TYPE,TRAN,SERIAL,PTUUID,START,SIZE,LOG-SEC'))['blockdevices']
    usb=[d for d in disks if d.get('tran')=='usb' and d.get('serial')=='BEAMOBOOT']
    assert len(usb)==1
    disk=usb[0]; parts=[p for p in disk['children'] if p['type']=='part']; assert len(parts)==1
    part=parts[0]; sector=int(disk['log-sec']); assert sector==512
    root=pathlib.Path('/media/beamo'); root.mkdir(parents=True,exist_ok=True)
    subprocess.run(['/usr/bin/mount','-o','ro',part['path'],str(root)],check=True)
    exe=root/'Start Beamo Wipe Linux'; assert exe.is_file()
    # Seed an exact existing boot entry only in this disposable virtual PC.
    # This models the documented supported firmware case; it is not product code.
    sig=struct.pack('<I',int(disk['ptuuid'],16))+bytes(12)
    hd=bytes([4,1])+struct.pack('<H',42)+struct.pack('<IQQ',1,int(part['start']),int(part['size'])//sector)+sig+bytes([1,1])
    name='\\EFI\\BOOT\\BOOTX64.EFI\0'.encode('utf-16-le')
    path=hd+bytes([4,4])+struct.pack('<H',4+len(name))+name+bytes([127,255,4,0])
    raw=struct.pack('<IH',1,len(path))+'Beamo test USB\0'.encode('utf-16-le')+path
    efidir=pathlib.Path('/sys/firmware/efi/efivars'); suffix='-8be4df61-93ca-11d2-aa0d-00e098032b8c'
    entry=efidir/('BootBEEF'+suffix); entry.write_bytes(struct.pack('<I',7)+raw)
    order=efidir/('BootOrder'+suffix); old=order.read_bytes(); assert len(old)>=6
    order.write_bytes(old+struct.pack('<H',0xBEEF))
    assert not (efidir/('BootNext'+suffix)).exists()
    proc=subprocess.Popen([str(exe)],stdout=subprocess.DEVNULL,stderr=subprocess.DEVNULL)
    urlfile=pathlib.Path('/run/beamo-browser-url')
    for _ in range(100):
        if urlfile.exists(): break
        assert proc.poll() is None
        time.sleep(.1)
    url=urllib.parse.urlsplit(urlfile.read_text().strip()); assert url.hostname=='127.0.0.1'
    origin=f'{url.scheme}://{url.netloc}'
    def api(action,confirm=False):
        req=urllib.request.Request(origin+'/api/'+action,data=json.dumps({'confirm':confirm}).encode(),headers={'Content-Type':'application/json','X-Beamo-Token':url.fragment,'Origin':origin},method='POST')
        with urllib.request.urlopen(req,timeout=30) as response:return json.load(response)
    view=api('check'); assert view['ready'], view
    marker('BEAMO_DESKTOP_CHECK_READY')
    marker('BEAMO_DESKTOP_RESTART_REQUESTED')
    try: api('restart',True)
    except Exception: pass  # A normal reboot can close the HTTP connection.
    time.sleep(60)
    raise RuntimeError('desktop did not reboot')
except Exception as exc:
    marker('BEAMO_DESKTOP_FAILURE='+type(exc).__name__+':'+str(exc)[:300])
    raise
